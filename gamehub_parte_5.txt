function setCarouselItem(index) {
    const featured = games.slice(0, 3);
    if (index < 0) index = featured.length - 1;
    if (index >= featured.length) index = 0;
    currentCarouselIndex = index;
    const main = document.getElementById('carousel-main');
    if (main) main.innerHTML = renderCarouselItem(featured[index], index);
    document.querySelectorAll('.carousel-dot').forEach((dot, i) => {
        dot.classList.toggle('active', i === index);
    });
}

function toggleCart() {
    document.getElementById('cart-sidebar').classList.toggle('open');
}

function addToCart(gameId) {
    const game = games.find(g => g.id === gameId);
    if (!game || cart.some(item => item.id === gameId)) return;
    cart.push(game);
    saveCart();
    updateCartUI();
    showToast(game.title + ' adicionado ao carrinho!', 'success');
}

function removeFromCart(gameId) {
    cart = cart.filter(item => item.id !== gameId);
    saveCart();
    updateCartUI();
    showToast('Item removido do carrinho', 'info');
    const storeGrid = document.getElementById('store-grid');
    if (storeGrid) {
        const activeFilter = document.querySelector('.filter-btn.active');
        const genre = activeFilter ? activeFilter.textContent : 'Todos';
        filterGames(genre === 'Todos' ? 'all' : genre === 'Game Pass' ? 'free' : genre, activeFilter);
    }
}

function saveCart() {
    localStorage.setItem('gamehub_cart', JSON.stringify(cart));
}

function updateCartUI() {
    const cartItems = document.getElementById('cart-items');
    const cartFooter = document.getElementById('cart-footer');
    const cartCount = document.getElementById('cart-count');
    cartCount.textContent = cart.length;
    if (cart.length === 0) {
        cartItems.innerHTML = `
            <div class="cart-empty">
                <div class="cart-empty-icon">🛒</div>
                <p>Seu carrinho está vazio</p>
                <p style="font-size: 0.9rem; margin-top: 0.5rem;">Adicione alguns jogos!</p>
            </div>
        `;
        cartFooter.style.display = 'none';
        return;
    }
    cartItems.innerHTML = cart.map(item => `
        <div class="cart-item">
            <div class="cart-item-image" style="background: ${item.bgGradient};"><span style="font-size: 2rem;">${item.image}</span></div>
            <div class="cart-item-info">
                <div class="cart-item-title">${escapeHtml(item.title)}</div>
                <div class="cart-item-price">${formatPrice(item.price)}</div>
            </div>
            <button class="cart-item-remove" onclick="removeFromCart(${item.id})">🗑️</button>
        </div>
    `).join('');
    cartFooter.style.display = 'block';
    updateCartTotals();
}

function updateCartTotals() {
    const subtotal = cart.reduce((sum, item) => sum + item.price, 0);
    const discountEl = document.getElementById('cart-discount');
    const discountRow = document.getElementById('cart-discount-row');
    let discount = 0;
    if (discountEl && discountEl.dataset.value) {
        discount = parseFloat(discountEl.dataset.value);
    }
    document.getElementById('cart-subtotal').textContent = formatPrice(subtotal);
    if (discountRow) discountRow.style.display = discount > 0 ? 'flex' : 'none';
    if (discountEl) discountEl.textContent = '- ' + formatPrice(discount);
    document.getElementById('cart-total').textContent = formatPrice(Math.max(0, subtotal - discount));
}

function applyCoupon() {
    const code = document.getElementById('coupon-code').value.trim().toUpperCase();
    const discountEl = document.getElementById('cart-discount');
    if (code === 'GAMEHUB10') {
        const subtotal = cart.reduce((sum, item) => sum + item.price, 0);
        discountEl.dataset.value = subtotal * 0.10;
        showToast('Cupom GAMEHUB10 aplicado! 10% OFF!', 'success');
    } else if (code === 'BEMVINDO') {
        const subtotal = cart.reduce((sum, item) => sum + item.price, 0);
        discountEl.dataset.value = subtotal * 0.20;
        showToast('Cupom BEMVINDO aplicado! 20% OFF!', 'success');
    } else {
        delete discountEl.dataset.value;
        showToast('Cupom inválido!', 'error');
    }
    updateCartTotals();
}

function handleLogin(e) {
    e.preventDefault();
    const email = document.getElementById('login-email').value;
    const password = document.getElementById('login-password').value;
    const btn = document.getElementById('login-btn');
    const btnText = document.getElementById('login-btn-text');
    document.querySelectorAll('.form-error').forEach(el => el.classList.remove('visible'));
    document.querySelectorAll('.form-input').forEach(el => el.classList.remove('error'));
    btn.disabled = true;
    btnText.textContent = 'ENTRANDO...';
    setTimeout(() => {
        const users = JSON.parse(localStorage.getItem('gamehub_users') || '[]');
        const user = users.find(u => u.email === email && u.password === btoa(password));
        if (!user) {
            btn.disabled = false;
            btnText.textContent = 'ENTRAR';
            document.getElementById('login-password').classList.add('error');
            document.getElementById('login-password-error').classList.add('visible');
            showToast('Email ou senha incorretos!', 'error');
            return;
        }
        currentUser = { id: user.id, name: user.name, email: user.email, username: user.username };
        localStorage.setItem('gamehub_user', JSON.stringify(currentUser));
        btn.disabled = false;
        btnText.textContent = 'ENTRAR';
        closeModal('auth-modal');
        updateAuthUI();
        showToast('Bem-vindo de volta, ' + user.name.split(' ')[0] + '!', 'success');
        document.getElementById('login-form').reset();
    }, 1000);
}

function handleRegister(e) {
    e.preventDefault();
    const name = document.getElementById('register-name').value;
    const username = document.getElementById('register-username').value;
    const email = document.getElementById('register-email').value;
    const password = document.getElementById('register-password').value;
    const confirm = document.getElementById('register-confirm').value;
    const btn = document.getElementById('register-btn');
    const btnText = document.getElementById('register-btn-text');
    document.querySelectorAll('.form-error').forEach(el => el.classList.remove('visible'));
    document.querySelectorAll('.form-input').forEach(el => el.classList.remove('error'));
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email)) {
        document.getElementById('register-email').classList.add('error');
        document.getElementById('register-email-error').classList.add('visible');
        return;
    }
    if (password.length < 8) {
        showToast('Senha deve ter pelo menos 8 caracteres!', 'error');
        document.getElementById('register-password').classList.add('error');
        return;
    }
    if (password !== confirm) {
        document.getElementById('register-confirm').classList.add('error');
        document.getElementById('register-confirm-error').classList.add('visible');
        return;
    }
    btn.disabled = true;
    btnText.textContent = 'CRIANDO CONTA...';
    setTimeout(() => {
        const users = JSON.parse(localStorage.getItem('gamehub_users') || '[]');
        if (users.some(u => u.email === email)) {
            btn.disabled = false;
            btnText.textContent = 'CRIAR CONTA';
            showToast('Este email já está cadastrado!', 'error');
            return;
        }
        if (users.some(u => u.username === username)) {
            btn.disabled = false;
            btnText.textContent = 'CRIAR CONTA';
            showToast('Este nome de usuário já existe!', 'error');
            return;
        }
        const newUser = {
            id: Date.now(), name, username, email,
            password: btoa(password),
            createdAt: new Date().toISOString()
        };
        users.push(newUser);
        localStorage.setItem('gamehub_users', JSON.stringify(users));
        currentUser = { id: newUser.id, name: newUser.name, email: newUser.email, username: newUser.username };
        localStorage.setItem('gamehub_user', JSON.stringify(currentUser));
        btn.disabled = false;
